## TASK 2 — Add claim token generation endpoint to server/routes.ts

Add a new route:
  POST /api/vendors/:id/generate-claim-link

Logic:
1. Look up the vendor by id (select id, businessName, claimToken, 
   claimTokenExpiresAt, claimStatus)
2. If a valid non-expired token already exists (claimTokenExpiresAt > now 
   AND claimStatus === 'unclaimed'), return the existing one — do not 
   regenerate.
3. Otherwise generate a new 64-character random token (use nanoid or 
   crypto.randomBytes(32).toString('hex')).
4. Set claimTokenExpiresAt = now + 7 days, claimStatus = 'unclaimed'.
5. Update the vendors row in Supabase.
6. Return:
   { claimUrl: `https://voomparts.com/claim/${token}`, token }

## TASK 6 — Add generateClaimLink to client/src/lib/voomApi.ts

export async function generateClaimLink(vendorId: number): 
  Promise<{ claimUrl: string; token: string }> {
  const res = await fetch(`/api/vendors/${vendorId}/generate-claim-link`, 
    { method: 'POST' });
  if (!res.ok) throw new Error('Failed to generate claim link');
  return res.json();
}
